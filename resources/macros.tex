\newcommand{\drawtriad}[1]{%
    \draw[vector guide, arrowred] (#1) -- (#11);
    \draw[vector guide, dashed, arrowgreen] (#1) -- (#12);
    \draw[vector guide, densely dotted, arrowblue] (#1) -- (#13);
}

    \draw[axis] (0.7, 0, 0) -- (4,0,0) node[anchor=north east] {$x$};
    \draw[axis] (0, 0.7, 0) -- (0,4,0) node[anchor=north west] {$y$};
    \draw[axis] (0, 0, 0.7) -- (0,0,4) node[anchor=south] {$z$};
}

\newcommand{\phantompathxy}{%
    \draw [draw=none, name path=xy] plot [smooth] coordinates {
        (0, 3.5, 0) (1.4, 3.4, 0) (2.6, 2.7, 0) (3.1, 1.9, 0) (3.5, 0, 0)
    };
}

\newcommand{\centerarc}[7]{%
  % #1 = center (e.g. C3)
  % #2 = radius
  % #3 = start angle
  % #4 = end angle
  % #5 = offset vector (dx,dy,dz)
  % #6 = node options
  % #7 = label
  \draw[-{Stealth[length=3.5pt, width=3pt]}]
    ($(#1) + (#5)$)
    arc[start angle=#3, end angle=#4, radius=#2]
    node[#6] {#7};
}

\newcommand{\drawplane}[5]{%
    % #1 = style/color
    % #2 = curved path name
    % #3 = curved path coordinates
    % #4 = axis path name
    % #5 = axis coordinates
    \draw[#1, name path=#2] plot [smooth] coordinates {#3};
    \draw[transparent, name path=#4] plot [smooth, tension=0] coordinates {#5};
    \tikzfillbetween[of=#2 and #4] {#1};
}
